function predictNeed(history, today):
  # history: array of {product, date}

  build map purchasesByProduct

  globalDowWeights = buildDowWeights(history)  # opcional

  results = []
  for each product p:
    dates = sort(purchasesByProduct[p])
    if len(dates) < 2: continue

    intervals = diffDays(dates)
    intervals = removeZero(intervals)
    intervals = clipOutliers(intervals, p10, p90)

    d = daysBetween(today, last(dates))

    if len(intervals) >= 3:
      (k, λ) = fitWeibullByMoments(intervals)
    else:
      (k, λ) = fallbackFromUserGlobal(history)  # p.ej. k=2, λ=mediana global

    p_base = 1 - exp(-(d/λ)^k)

    m_dow = dowMultiplier(dates, today)  # Laplace smoothing; si no hay datos usa globalDowWeights o 1
    p_season = clamp(p_base * m_dow, 0, 1)

    expected = λ * Gamma(1 + 1/k)
    r = d / expected

    churnPenalty = 1
    if d > 180 OR r > 3:
      churnPenalty = exp(-0.45*(r-1))

    p_final = p_season * churnPenalty

    conf = computeConfidence(intervals)
    p_scored = p_final * (0.5 + 0.5*conf)

    results.push({
      product: p,
      p_need: p_scored,
      expected_next_date: last(dates) + expected days,
      days_overdue: d - expected,
      confidence: conf,
      reason: explain(p, d, expected, conf)
    })

  sort results by p_need desc
  return results
